\textbf{Lemma 4} \quad I_k(a,b) \geq x \quad \textit{iff} \quad \sum_i \min\bigl(I_{i,k}(a,b),x\bigr) \geq mx.

\bigskip

\noindent
\textbf{Proof.}

\bigskip

\noindent
\textit{Only If:} Let $\xi$ be the number of tasks for which it is $I_{i,k}(a,b) \geq x$.
\[
\sum_{i=1}^{n} \min\bigl(I_{i,k}(a,b),x\bigr)
=
\xi x + \sum_{i:I_{i,k}<x} I_{i,k}(a,b)
=
\]
\[
=
\xi x + mI_k(a,b) - \sum_{i:I_{i,k}\geq x} I_{i,k}(a,b)
\geq
\]
\[
\geq
\xi x + mI_k(a,b) - \xi I_k(a,b)
\geq
\xi x + (m-\xi)x
=
mx.
\]

\noindent
\textit{If:} Note that if $\sum_i \min\bigl(I_{i,k}(a,b),x\bigr) \geq mx$, it follows that
\[
I_k(a,b)
=
\sum_i \frac{I_{i,k}(a,b)}{m}
\geq
\sum_i \frac{\min\bigl(I_{i,k}(a,b),x\bigr)}{m}
\geq
\frac{mx}{m}
=
x
\]
