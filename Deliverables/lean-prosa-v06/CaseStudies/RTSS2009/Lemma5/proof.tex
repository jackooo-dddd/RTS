\textbf{Lemma 5.} \ \textit{Given a $t_0 = r_k^1 - \varphi$, let for each $h \geq 1$ denote $\chi^h$ the minimal solution of the recursive equation}
\[
x=\left\lceil \frac{\Omega_k(x,h)}{M} \right\rceil + h \cdot C_k.
\tag{16}
\]

\textit{Let further $H(\varphi)$ be the minimal index satisfying}
\[
\chi^{H(\varphi)} \leq H(\varphi)\cdot T_k + \varphi.
\tag{17}
\]

\textit{Then}
\[
R_k^\varphi=\max_{h\in[1,H(\varphi)]}\left\{\chi^h-(h-1)\cdot T_k-\varphi\right\}
\tag{18}
\]

\textit{is an upper bound of $\tau_k$'s response time with this particular $t_0$.}
